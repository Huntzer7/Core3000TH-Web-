## script.js

```javascript
// =========================
// GLOBAL VARIABLES
// =========================

let words = [];

let currentIndex = Number(localStorage.getItem("currentIndex")) || 0;
let learnedWords = JSON.parse(localStorage.getItem("learnedWords")) || [];

let quizWords = [];
let quizIndex = 0;
let correct = 0;
let incorrect = 0;

// =========================
// PAGE ELEMENTS
// =========================

const homePage = document.getElementById("homePage");
const learningPage = document.getElementById("learningPage");
const quizPage = document.getElementById("quizPage");
const resultPage = document.getElementById("resultPage");

// HOME
const learnedCount = document.getElementById("learnedCount");
const remainingCount = document.getElementById("remainingCount");

// LEARNING
const wordText = document.getElementById("word");
const wordType = document.getElementById("wordType");
const meaning = document.getElementById("meaning");
const meaningBtn = document.getElementById("meaningBtn");
const meaningBox = document.getElementById("meaningBox");
const wordProgress = document.getElementById("wordProgress");
const remainingWords = document.getElementById("remainingWords");

// QUIZ
const quizWord = document.getElementById("quizWord");
const quizChoices = document.getElementById("quizChoices");
const quizResult = document.getElementById("quizResult");

// RESULT
const correctCount = document.getElementById("correctCount");
const incorrectCount = document.getElementById("incorrectCount");
const scorePercent = document.getElementById("scorePercent");

// =========================
// BUTTONS
// =========================

const startBtn = document.getElementById("startBtn");
const nextBtn = document.getElementById("nextBtn");
const backBtn = document.getElementById("backBtn");
const modeBtn = document.getElementById("modeBtn");
const restartBtn = document.getElementById("restartBtn");
const newWordBtn = document.getElementById("newWordBtn");

// =========================
// FETCH WORD DATA
// =========================

fetch("words.json")
  .then((response) => response.json())
  .then((data) => {
    words = data;

    updateHome();
    showWord();
  });

// =========================
// PAGE SWITCHING
// =========================

function showPage(page) {
  [homePage, learningPage, quizPage, resultPage].forEach((p) => {
    p.classList.remove("active");
  });

  page.classList.add("active");
}

// =========================
// HOME UPDATE
// =========================

function updateHome() {
  learnedCount.textContent = `Learned: ${learnedWords.length} words`;
  remainingCount.textContent = `Remaining: ${words.length - learnedWords.length} words`;
}

// =========================
// SHOW CURRENT WORD
// =========================

function showWord() {
  const currentWord = words[currentIndex];

  wordText.textContent = currentWord.word;
  wordType.textContent = currentWord.type;
  meaning.textContent = currentWord.meaning;

  wordProgress.textContent = `Word ${currentIndex + 1} / ${words.length}`;
  remainingWords.textContent = `Remaining: ${words.length - learnedWords.length}`;

  meaningBox.classList.add("hidden");
  meaningBtn.textContent = "Show Meaning";
}

// =========================
// SHOW/HIDE MEANING
// =========================

meaningBtn.addEventListener("click", () => {
  meaningBox.classList.toggle("hidden");

  if (meaningBox.classList.contains("hidden")) {
    meaningBtn.textContent = "Show Meaning";
  } else {
    meaningBtn.textContent = "Hide Meaning";
  }
});

// =========================
// START LEARNING
// =========================

startBtn.addEventListener("click", () => {
  showPage(learningPage);
  showWord();
});

// =========================
// NEXT WORD
// =========================

nextBtn.addEventListener("click", () => {

  if (!learnedWords.includes(currentIndex)) {
    learnedWords.push(currentIndex);
  }

  localStorage.setItem("learnedWords", JSON.stringify(learnedWords));

  if (currentIndex < words.length - 1) {
    currentIndex++;
  }

  localStorage.setItem("currentIndex", currentIndex);

  showWord();
  updateHome();
});

// =========================
// PREVIOUS WORD
// =========================

backBtn.addEventListener("click", () => {
  if (currentIndex > 0) {
    currentIndex--;
    localStorage.setItem("currentIndex", currentIndex);
    showWord();
  }
});

// =========================
// QUIZ MODE
// =========================

modeBtn.addEventListener("click", () => {

  if (learnedWords.length === 0) {
    alert("Learn some words first!");
    return;
  }

  quizWords = learnedWords.map(index => words[index]);

  quizIndex = 0;
  correct = 0;
  incorrect = 0;

  showPage(quizPage);
  showQuiz();
});

// =========================
// SHOW QUIZ QUESTION
// =========================

function showQuiz() {

  quizResult.textContent = "";

  const currentQuiz = quizWords[quizIndex];

  quizWord.textContent = currentQuiz.word;

  let choices = [currentQuiz.meaning];

  while (choices.length < 4) {
    const randomWord = words[Math.floor(Math.random() * words.length)];

    if (!choices.includes(randomWord.meaning)) {
      choices.push(randomWord.meaning);
    }
  }

  choices = choices.sort(() => Math.random() - 0.5);

  quizChoices.innerHTML = "";

  choices.forEach(choice => {

    const button = document.createElement("button");
    button.classList.add("choiceBtn");
    button.textContent = choice;

    button.addEventListener("click", () => {
      checkAnswer(choice, currentQuiz.meaning);
    });

    quizChoices.appendChild(button);
  });
}

// =========================
// CHECK ANSWER
// =========================

function checkAnswer(selected, correctAnswer) {

  const buttons = document.querySelectorAll(".choiceBtn");
  buttons.forEach(btn => btn.disabled = true);

  if (selected === correctAnswer) {
    quizResult.textContent = "Correct";
    correct++;
  } else {
    quizResult.textContent = `Incorrect | Correct Answer: ${correctAnswer}`;
    incorrect++;
  }

  setTimeout(() => {

    quizIndex++;

    if (quizIndex >= quizWords.length) {
      showResult();
    } else {
      showQuiz();
    }

  }, 3000);
}

// =========================
// RESULT PAGE
// =========================

function showResult() {

  showPage(resultPage);

  const total = correct + incorrect;
  const score = Math.round((correct / total) * 100);

  correctCount.textContent = `Correct: ${correct}`;
  incorrectCount.textContent = `Incorrect: ${incorrect}`;
  scorePercent.textContent = `Score: ${score}%`;
}

// =========================
// RESTART QUIZ
// =========================

restartBtn.addEventListener("click", () => {

  quizIndex = 0;
  correct = 0;
  incorrect = 0;

  showPage(quizPage);
  showQuiz();
});

// =========================
// RETURN TO LEARNING
// =========================

newWordBtn.addEventListener("click", () => {
  showPage(learningPage);
  showWord();
});
```

# HOW TO RUN

1. Create a folder

2. Put these 4 files inside:

   * index.html
   * style.css
   * script.js
   * words.json

3. Open index.html in browser

---

# HOW TO DEPLOY TO GITHUB PAGES

1. Create repository on GitHub
2. Upload all files
3. Go to:
   Settings → Pages
4. Select:
   Deploy from branch
5. Choose:
   main branch
6. Save

GitHub will generate your website URL.
